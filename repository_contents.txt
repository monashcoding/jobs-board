package main

import (
	"flag"
	"fmt"
	"io"
	"os"
	"path/filepath"
	"strings"
)

// Config holds the script configuration
type Config struct {
	ignoredExtensions map[string]bool
	ignoredFiles      map[string]bool
}

func main() {
	// Define flags
	sourcePath := flag.String("source", ".", "Source directory path")
	outputPath := flag.String("output", "repository_contents.txt", "Output file path")
	ignoreExts := flag.String("ignore-ext", ".git,.gitignore,.idea,node_modules,.vscode,.env,.txt", "Comma-separated list of extensions/directories to ignore")
	ignoreFiles := flag.String("ignore-files", "go.sum,cmd/repo2text/main.go,README.md,.DS_Store", "Comma-separated list of specific files to ignore")
	flag.Parse()

	// Initialize config
	config := &Config{
		ignoredExtensions: make(map[string]bool),
		ignoredFiles:      make(map[string]bool),
	}

	// Process ignored extensions
	for _, ext := range strings.Split(*ignoreExts, ",") {
		ext = strings.TrimSpace(ext)
		if ext != "" {
			config.ignoredExtensions[ext] = true
		}
	}

	// Process ignored files
	for _, file := range strings.Split(*ignoreFiles, ",") {
		file = strings.TrimSpace(file)
		if file != "" {
			config.ignoredFiles[file] = true
		}
	}

	// Create output file
	outputFile, err := os.Create(*outputPath)
	if err != nil {
		fmt.Printf("Error creating output file: %v\n", err)
		os.Exit(1)
	}
	defer outputFile.Close()

	// Process the repository
	err = processRepository(*sourcePath, outputFile, config)
	if err != nil {
		fmt.Printf("Error processing repository: %v\n", err)
		os.Exit(1)
	}

	fmt.Printf("Repository contents written to %s\n", *outputPath)
}

func processRepository(sourcePath string, output io.Writer, config *Config) error {
	return filepath.Walk(sourcePath, func(path string, info os.FileInfo, err error) error {
		if err != nil {
			return err
		}

		// Skip if it's a directory
		if info.IsDir() {
			// Check if directory should be ignored
			if config.shouldIgnore(info.Name()) {
				return filepath.SkipDir
			}
			return nil
		}

		// Check if file should be ignored
		if config.shouldIgnore(info.Name()) {
			return nil
		}

		// Read file contents
		content, err := os.ReadFile(path)
		if err != nil {
			return fmt.Errorf("error reading file %s: %w", path, err)
		}

		// Write file path and contents to output
		relPath, err := filepath.Rel(sourcePath, path)
		if err != nil {
			return fmt.Errorf("error getting relative path for %s: %w", path, err)
		}

		// Write file header and contents
		_, err = fmt.Fprintf(output, "// File: %s\n\n%s\n\n", relPath, string(content))
		if err != nil {
			return fmt.Errorf("error writing to output file: %w", err)
		}

		return nil
	})
}

func (c *Config) shouldIgnore(name string) bool {
	// Check if the file/directory is in the ignored files list
	if c.ignoredFiles[name] {
		return true
	}

	// Check if the file/directory extension is in the ignored extensions list
	ext := filepath.Ext(name)
	if c.ignoredExtensions[ext] {
		return true
	}

	// Check if the entire name matches any ignored extension
	// This handles directories like .git, node_modules, etc.
	if c.ignoredExtensions[name] {
		return true
	}

	return false
}

package main

import (
	"context"
	"os"
	"os/signal"
	"syscall"

	"github.com/monashcoding/mploy-golang-core/internal/models"
	"github.com/monashcoding/mploy-golang-core/internal/pipeline/steps"
	"github.com/monashcoding/mploy-golang-core/internal/utils"

	"github.com/monashcoding/mploy-golang-core/internal/database"
	"github.com/monashcoding/mploy-golang-core/internal/pipeline"
	"github.com/monashcoding/mploy-golang-core/internal/scraper"
	"github.com/monashcoding/mploy-golang-core/internal/scraper/sites"
)

func main() {
	utils.Log("Starting MPLOY Scraper.")

	// Set up context with cancellation
	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()

	// Set up MongoDB connection
	utils.Log("Setting up MongoDB connection.")
	mongodb, err := database.NewMongoDB(ctx)
	if err != nil {
		utils.Fatal("Failed to connect to MongoDB: %v", err)
	}
	defer mongodb.Disconnect(ctx)

	// Set up graceful shutdown
	shutdown := make(chan os.Signal, 1)
	signal.Notify(shutdown, syscall.SIGINT, syscall.SIGTERM)
	go func() {
		<-shutdown
		utils.Log("Received shutdown signal, shutting down...")
		cancel()
	}()

	// Initialize scrapers
	scrapers := []scraper.Scraper{
		//sites.NewGradconScraper(),
		sites.NewProspleScraper(),
		//sites.NewLinkedinScraper(),
	}

	// Initialize pipeline
	processingPipeline := pipeline.NewPipeline(
		steps.NewLogStep("Initial"),
		//steps.NewNormalizationStep(),
		//steps.NewDeduplicationStep(),
		steps.NewMongoDBStep(mongodb),
	)

	// Collect all listings from scrapers
	utils.Log("Scraping started...")

	var allListings []models.Listing
	for _, s := range scrapers {
		utils.Log("Running scraper %s...", s.Name())
		listings, err := s.Scrape(ctx)
		if err != nil {
			utils.Error("[X] Error in %s scraper: %v", s.Name(), err)
			continue
		}
		utils.Log("Finished scraper %s with %d listings.", s.Name(), len(listings))
		allListings = append(allListings, listings...)
	}

	utils.Log("Scraping section finished [\u2713]")
	utils.Log("Starting pipeline section [...]")

	// Run pipeline with scrapers
	if _, err := processingPipeline.Run(ctx, allListings); err != nil {
		utils.Error("Error processing listings: %v", err)
	}

	utils.Log("Pipeline section finished [\u2713]")
	
	utils.Log("Processing completed [\u2713]")
}

type MongoDB struct {

func init() {
	utils.Log("Loading environment variables from .env file...")
	if err := godotenv.Load(); err != nil {
		utils.Fatal(".env file not found: %v\n", err)
	}
	utils.Log("Environment variables loaded.")
}

// NewMongoDB creates a new MongoDB instance
func NewMongoDB(ctx context.Context) (*MongoDB, error) {
	// Configure client options
	clientOptions := options.Client().
		ApplyURI(os.Getenv("MONGODB_URI")).
		SetConnectTimeout(10 * time.Second).
		SetMaxConnIdleTime(60 * time.Second).
		SetMinPoolSize(5).
		SetMaxPoolSize(100)

	// Create client
	client, err := mongo.Connect(ctx, clientOptions)
	if err != nil {
		return nil, fmt.Errorf("failed to create MongoDB client: %w", err)
	}

	// Ping database to verify connection
	ctxPing, cancel := context.WithTimeout(ctx, 5*time.Second)
	defer cancel()

	if err := client.Ping(ctxPing, readpref.Primary()); err != nil {
		return nil, fmt.Errorf("failed to ping MongoDB: %w", err)
	}

	// Get database and collection
	database := client.Database(os.Getenv("MONGODB_DATABASE"))
	collection := database.Collection(os.Getenv("MONGODB_COLLECTION"))

	return &MongoDB{
		client:     client,
		database:   database,
		collection: collection,
	}, nil
}

// GetCollection returns the MongoDB collection
func (m *MongoDB) GetCollection() *mongo.Collection {
	return m.collection
}

// Disconnect closes the MongoDB connection
func (m *MongoDB) Disconnect(ctx context.Context) error {
	if err := m.client.Disconnect(ctx); err != nil {
		return fmt.Errorf("failed to disconnect from MongoDB: %w", err)
	}
	return nil
}

func (m *MongoDB) GetListings() ([]models.Listing, error) {
	var listings []models.Listing
	cursor, err := m.collection.Find(context.Background(), bson.D{}) // Add filter options to bson.D object.

	if err != nil {
		return listings, fmt.Errorf("failed to get all listings: %w", err)
	}

	defer cursor.Close(context.Background())

	for cursor.Next(context.Background()) {
		var listing models.Listing
		if err := cursor.Decode(&listing); err != nil {
			return listings, fmt.Errorf("failed to decode listing: %w", err)
		}
		listings = append(listings, listing)
	}

	return listings, nil
}

package pipeline

import (
	"context"
	"fmt"
	"github.com/monashcoding/mploy-golang-core/internal/models"
	"github.com/monashcoding/mploy-golang-core/internal/utils"
)

// Step represents a single processing step in the pipeline
type Step interface {
	Name() string
	Process(ctx context.Context, listings []models.Listing) ([]models.Listing, error)
}

// Pipeline manages the processing steps
type Pipeline struct {
	steps []Step
}

func NewPipeline(steps ...Step) *Pipeline {
	return &Pipeline{
		steps: steps,
	}
}

// Run executes all pipeline steps in order
func (p *Pipeline) Run(ctx context.Context, listings []models.Listing) ([]models.Listing, error) {
	var err error
	currentListings := listings

	for _, step := range p.steps {
		utils.Log("Running step: " + step.Name())
		currentListings, err = step.Process(ctx, currentListings)
		if err != nil {
			return nil, fmt.Errorf("error in %s step: %w", step.Name(), err)
		}
		utils.Log("Finished step: " + step.Name())
	}

	return currentListings, nil
}

package steps

import (
	"context"
	"github.com/monashcoding/mploy-golang-core/internal/models"
	"github.com/monashcoding/mploy-golang-core/internal/utils"
	"strings"
)

type LogStep struct {
	prefix string
}

func NewLogStep(prefix string) *LogStep {
	return &LogStep{
		prefix: prefix,
	}
}

func (s *LogStep) Name() string {
	return "Log"
}

func (s *LogStep) Process(ctx context.Context, listings []models.Listing) ([]models.Listing, error) {
	utils.Log("\n=== %s: Processing %d listings ===", s.prefix, len(listings))

	for _, listing := range listings {
		// Create a compact location string
		locations := strings.Join(compactStringSlice(locationsToStrings(listing.Locations)), "/")

		// Get company name or "Unknown"
		company := listing.Company.Name
		if company == "" {
			company = "Unknown"
		}

		// Format close date if exists
		closeDate := "No close date"
		if listing.CloseDate != nil {
			closeDate = listing.CloseDate.Format("2006-01-02")
		}

		// Create compact string for study fields
		fields := strings.Join(compactStringSlice(studyFieldsToStrings(listing.StudyFields)), "/")
		if fields == "" {
			fields = "Any"
		}

		// Print compact single-line summary
		utils.Log("[%s] %s @ %s | Type: %s | Loc: %s | Fields: %s | Closes: %s | URL: %s",
			listing.ID.Hex()[:8],
			truncateString(listing.Title, 40),
			company,
			listing.Type,
			locations,
			fields,
			closeDate,
			truncateString(listing.ApplicationURL, 40),
		)
	}
	return listings, nil
}

// Helper functions

func truncateString(s string, length int) string {
	if len(s) <= length {
		return s
	}
	return s[:length-3] + "..."
}

func compactStringSlice(slice []string) []string {
	if len(slice) <= 3 {
		return slice
	}
	return append(slice[:2], "...")
}

func locationsToStrings(locs []models.LocationType) []string {
	result := make([]string, len(locs))
	for i, loc := range locs {
		result[i] = string(loc)
	}
	return result
}

func studyFieldsToStrings(fields []models.StudyField) []string {
	result := make([]string, len(fields))
	for i, field := range fields {
		result[i] = string(field)
	}
	return result
}

	if len(got) != len(want) {
		t.Errorf("Got %d listings, want %d", len(got), len(want))
		return
	}

		gotJSON, err := json.Marshal(got[i])
		if err != nil {
			t.Fatalf("Failed to marshal got listing: %v", err)

		wantJSON, err := json.Marshal(want[i])
		if err != nil {
			t.Fatalf("Failed to marshal want listing: %v", err)
		}

		if string(gotJSON) != string(wantJSON) {

		}
	}

func OpportunityToModel(baseURL string, opportunity Opportunity) (*models.Listing, error) {
	sourceUrl := baseURL + opportunity.DetailPageURl

	company := models.Company{
		Name:    opportunity.ParentEmployer.Title,
		Website: baseURL + opportunity.ParentEmployer.DetailPageUrl,
	}

	jobType := models.JobType(opportunity.OpportunityTypes[0].Label)

	parsedCloseDate, err := time.Parse("2006-01-02T15:04:05.000Z", opportunity.ApplicationsCloseDate)
	if err != nil {
		return nil, fmt.Errorf("error parsing close date: %v", err)
	}

	locations, err := ProspleLocationsToLocationTypes(opportunity)
	if err != nil {
		return nil, fmt.Errorf("error getting locations: %v", err)
	}

	studyfield, err := ProspleStudyFieldsToStudyFields(opportunity)
	if err != nil {
		return nil, fmt.Errorf("error getting study fields: %v", err)
	}

	var parsedStartDate time.Time
	if opportunity.StartDate.Category != nil {
		parsedStartDate, err = time.Parse("2006-01-02T15:04:05.000Z", opportunity.StartDate.Category.Label)
		if err != nil {
			return nil, fmt.Errorf("error parsing start date: %v", err)
		}
	} else {
		parsedStartDate = time.Now()
	}

	workingRights, err := ProspleWorkingRightsToWorkingRights(opportunity)
	if err != nil {
		return nil, fmt.Errorf("error getting working rights: %v", err)
	}

	return &models.Listing{
		Title:          opportunity.Title,
		Description:    opportunity.Overview.FullText,
		Company:        company,
		ApplicationURL: opportunity.ApplyByUrl,
		SourceURLs:     []string{sourceUrl},
		Type:           jobType,
		CloseDate:      &parsedCloseDate,
		Locations:      locations,
		StudyFields:    studyfield,
		StartDate:      parsedStartDate,
		WorkingRights:  workingRights,
		CreatedAt:      time.Now(),
		UpdatedAt:      time.Now(),
	}, nil
}

func ProspleLocationsToLocationTypes(opportunity Opportunity) ([]models.LocationType, error) {
	locations := []models.LocationType{}
	for _, location := range opportunity.PhysicalLocations {
		locations = append(locations, models.LocationType(location.Label))

		for _, childLocation := range location.Children {
			locations = append(locations, models.LocationType(childLocation.Label))
		}
	}
	return locations, nil
}

func ProspleStudyFieldsToStudyFields(opportunity Opportunity) ([]models.StudyField, error) {
	studyFields := []models.StudyField{}
	for _, field := range opportunity.StudyFields {
		studyFields = append(studyFields, models.StudyField(field.Label))
	}
	return studyFields, nil
}

func ProspleWorkingRightsToWorkingRights(opportunity Opportunity) ([]models.WorkingRight, error) {
	workingRights := []models.WorkingRight{}
	for _, workingRight := range opportunity.WorkingRights {
		workingRights = append(workingRights, models.WorkingRight(workingRight.Label))

		for _, child := range workingRight.Children {
			workingRights = append(workingRights, models.WorkingRight(child.Label))
		}
	}
	return workingRights, nil
}

func SendGraphQLQuery(url string, query string, variables map[string]interface{}) (*GraphQLResponse, error) {
	requestBody := GraphQLRequest{
		Query:     query,
		Variables: variables,
	}

	utils.Log("Sending GraphQL request [...]")

	jsonBody, err := json.Marshal(requestBody)
	if err != nil {
		utils.Log("Failed to marshal request body: %v", err)
		return nil, fmt.Errorf("failed to marshal request body: %v", err)
	}

	req, err := http.NewRequest("POST", url, bytes.NewBuffer(jsonBody))
	if err != nil {
		utils.Log("Failed to create request: %v", err)
		return nil, fmt.Errorf("failed to create request: %v", err)
	}

	req.Header.Set("Content-Type", "application/json")

	client := &http.Client{}
	resp, err := client.Do(req)
	if err != nil {
		return nil, fmt.Errorf("failed to send request: %v", err)
	}
	defer resp.Body.Close()

	body, err := io.ReadAll(resp.Body)
	if err != nil {
		return nil, fmt.Errorf("failed to read response body: %v", err)
	}

	if resp.StatusCode != http.StatusOK {
		return nil, fmt.Errorf("HTTP error: %s, %v", resp.Status, string(body))
	}

	utils.Log("GraphQL response successfully received [\u2713]")

	var graphqlResp GraphQLResponse
	if err := json.Unmarshal(body, &graphqlResp); err != nil {
		return nil, fmt.Errorf("failed to unmarshal response: %v", err)
	}

	if len(graphqlResp.Errors) > 0 {
		return nil, fmt.Errorf("GraphQL error: %v", graphqlResp.Errors)
	}

	utils.Log("Reponse body successfully parsed [\u2713]")

	return &graphqlResp, nil
}

func (s *ProspleScraper) PerformGraphQLScrape(offset int, pageSize int) ([]models.Listing, error)  {
	prospleApiUrl := s.API_URL()
	var listings []models.Listing

	query := prosple.GetGraphQlQuery()
	variables := prosple.GetGraphQlVariables(offset, pageSize)

	response, err := prosple.SendGraphQLQuery(prospleApiUrl, query, variables)

	if err != nil {

	}

	for _, opp := range response.Data.OpportunitiesSearch.Opportunities {
		listing, err := prosple.OpportunityToModel(s.URL(), opp)
		if err != nil {
			fmt.Printf("Error converting opportunity to model: %v\n", err)
			continue
		}
		listings = append(listings, *listing)
	}

	return listings, nil
}

func (s *ProspleScraper) Scrape(ctx context.Context) ([]models.Listing, error) {
	var listings []models.Listing
	pages := 2 // Number of pages to scrape
	pageSize := 50 // Number of listings per page

	// Scrape all listings
	scrapeAll := true 
	if scrapeAll {
		// Scrape until pages are exhausted (i.e. the size of the returned array is not the same as the page size)
		for i := 0; ; i++ {
			utils.Log("Fully Scraping Prosple: Page %d", i + 1)

			newListings, err := s.PerformGraphQLScrape( i * pageSize, pageSize)

			if (err != nil) {
				return nil, fmt.Errorf("failed to scrape Prosple: %v", err)
			}

			if len(newListings) != pageSize {
				// End of results
				listings = append(listings, newListings...)
				break;
			}

			listings = append(listings, newListings...)
		}
		
		return listings, nil
	}


	// Scrape a fixed number of pages
	for i := 0; i < pages; i++ {		
		utils.Log("Scraping Prosple: Page %d of %d...", i + 1, pages)

		newListings, err := s.PerformGraphQLScrape(i * pageSize, pageSize)

		if (err != nil) {
			return nil, fmt.Errorf("failed to scrape Prosple: %v", err)
		}

		listings = append(listings, newListings...)
	}

func (m *mockTransport) RoundTrip(req *http.Request) (*http.Response, error) {
	// Get the fixture file for this URL
	fixtureFile, exists := m.fixtureMapping[req.URL.String()]
	if !exists {
		m.t.Fatalf("No fixture mapping found for URL: %s", req.URL.String())
	}

	// Load the fixture

	content, err := os.ReadFile(filename)
	if err != nil {
		m.t.Fatalf("Failed to read test fixture %s: %v", filename, err)
	}

	// Create mock response
	return &http.Response{
		StatusCode: 200,
		Body:       io.NopCloser(bytes.NewReader(content)),
		Header:     make(http.Header),
	}, nil
}

		if baseScraper, ok := s.Scraper.(interface{ SetCollector(*colly.Collector) }); ok {

// compareResults compares scraped listings with expected listings
func (s *ScraperTestSuite) compareResults(t *testing.T, got, want []models.Listing) {
	t.Helper()

	if len(got) != len(want) {
		t.Errorf("Got %d listings, want %d", len(got), len(want))
		return
	}

	for i := range got {
		s.compareListings(t, got[i], want[i])
	}
}

// compareListings compares two listings for equality
func (s *ScraperTestSuite) compareListings(t *testing.T, got, want models.Listing) {
	t.Helper()

	gotJSON, err := json.Marshal(got)
	if err != nil {
		t.Fatalf("Failed to marshal got listing: %v", err)
	}

	wantJSON, err := json.Marshal(want)
	if err != nil {
		t.Fatalf("Failed to marshal want listing: %v", err)
	}

	if string(gotJSON) != string(wantJSON) {
		t.Errorf("Listings don't match:\ngot:  %s\nwant: %s", gotJSON, wantJSON)
	}
}

package utils

import (
	"fmt"
	"os"
	"runtime"
	"strings"
	"time"
)

// Log prints a standard log message with optional formatting
func Log(format string, args ...interface{}) {
	message := fmt.Sprintf(format, args...)
	logWithOptionalLevel(message, "", 2)
}

// Error logs a message with ERROR level and optional formatting
func Error(format string, args ...interface{}) {
	message := fmt.Sprintf(format, args...)
	logWithOptionalLevel(message, "ERROR", 2)
}

// Debug logs a message with DEBUG level and optional formatting
func Debug(format string, args ...interface{}) {
	message := fmt.Sprintf(format, args...)
	logWithOptionalLevel(message, "DEBUG", 2)
}

// Fatal logs a message and terminates the program
func Fatal(format string, args ...interface{}) {
	message := fmt.Sprintf(format, args...)
	logWithOptionalLevel(message, "FATAL", 2)
	os.Exit(1)
}

func logWithOptionalLevel(message string, level string, skip int) {
	// Get the caller's function name
	pc, _, _, ok := runtime.Caller(skip)
	functionName := "unknown"
	if ok {
		fn := runtime.FuncForPC(pc)
		if fn != nil {
			// Get full function name
			fullName := fn.Name()
			// Split by "/" and get the last two parts (package.function)
			parts := strings.Split(fullName, "/")
			functionName = parts[len(parts)-1]
		}
	}

	// Format timestamp
	timestamp := time.Now().Format("15:04:05")

	if level != "" {
		fmt.Printf("[%s %s %s] [%s] %s\n", level, functionName, timestamp, message)
	} else {
		fmt.Printf("[%s %s] %s\n", functionName, timestamp, message)
	}
}
