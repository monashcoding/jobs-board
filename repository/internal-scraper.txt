	Name() string
	URL() string // Each scraper defines its own URLs

}

// CollyScraper provides common functionality for all scrapers
type CollyScraper struct {
	collector *colly.Collector
}

// NewCollyScraper creates a new base scraper with configured Colly collector
func NewCollyScraper() *CollyScraper {
	c := colly.NewCollector(
		colly.UserAgent("MPLOY"),
		colly.MaxDepth(2),
		colly.Async(true),
	)

	// Set up rate limiting
	c.Limit(&colly.LimitRule{
		DomainGlob:  "*",
		RandomDelay: 5 * time.Second,
		Parallelism: 2,
	})

	return &CollyScraper{
		collector: c,
	}
}

func (s* GradconScraper) PerformAPIScrape(offset int, pageSize int) ([]models.Listing, error) {
	gradConnectApiUrl := s.API_URL()
	var listings []models.Listing

	result, err := gradcon.SendAPIRequest(gradConnectApiUrl, pageSize, offset, s.unblocker)

	if err != nil {
		utils.Error("Failed to send API Request")
	}

	for _, item := range *result {
		gradConnectListings, err := gradcon.GradConnectToMPLOY(item, s.URL())

		if err != nil {
			fmt.Printf("Error converting opportunity to model: %v\n", err)
			continue
		}
		listings = append(listings, gradConnectListings...)
	}

	return listings, nil
}

func (s *GradconScraper) Scrape(ctx context.Context) ([]models.Listing, error) {
	pages := 2 // Number of pages to scrape
	pageSize := 50 // Number of listings per page

	listings := []models.Listing{}
	
	// Scrape all listings
	scrapeAll := false
	if scrapeAll {
		// Scrape until pages are exhausted (i.e. the size of the returned array is not the same as the page size)
		for i := 0; ; i++ {
			utils.Log("Fully Scraping GradConnect: Page %d", i + 1)

			newListings, err := s.PerformAPIScrape(i * pageSize, pageSize)

			if (err != nil) {
				return nil, fmt.Errorf("failed to scrape GradConnect: %v", err)
			}

			if len(newListings) != pageSize {
				// End of results
				listings = append(listings, newListings...)
				break;
			}

			listings = append(listings, newListings...)
		}
		
		return listings, nil
	}

	for i := 0; i < pages; i++ {
		utils.Log("Scraping GradConnect: Page %d of %d...", i + 1, pages)

		newListings, err := s.PerformAPIScrape(i * pageSize, pageSize)

		if (err != nil ){
			return nil, fmt.Errorf("failed to scrape GradConnect: %v", err)
		}

		listings = append(listings, newListings...)
	}

	return listings, nil
}

type APIResponse = []GradConnectData

type GradConnectData struct {
	Campaigns []GradConnectCampaign `json:"campaigns"`
	CustomerOrganisation GradConnectCustomerOrganisation `json:"customer_organization"`
	EarliestClosingDate string `json:"earliest_closing_date"`
}

type GradConnectCampaign struct {
	Id string `json:"id"`
	Title string `json:"title"`
	Slug string `json:"slug"`
	Description string `json:"description"`
	JobType string `json:"job_type"`	
	TargetURL string `json:"target_url"`
	Interval struct {
		Start string `json:"start"`
		End string `json:"end"`
	} `json:"interval"`
	Disciplines []string `json:"disciplines"`
	Locations []string `json:"locations"`
	WorkRights []string `json:"work_rights"`
	OriginTargetURL string `json:"origin_target_url"`
	AdditionalFields []GradConnectAdditionalField `json:"additional_fields"`
}

type GradConnectAdditionalField struct {
	Type  string      `json:"type"`
	Value interface{} `json:"value"`
}

type GradConnectCustomerOrganisation struct {
	Id string `json:"id"`
	Name string `json:"name"`
	Slug string `json:"slug"`
}

func SendAPIRequest(url string, limit int, offset int, unblocker unblocker.Service) (*APIResponse, error) {
	params := "?disciplines=computer-science&limit=20&limit=" + strconv.Itoa(limit) + "&offset=" + strconv.Itoa(offset) + "&recommended=true"
	apiUrl := url + params

	// Create a new request
	req, err := http.NewRequest("GET", apiUrl, nil)

	if err != nil {
		log.Fatal("Error creating request:", err)
	}

	// Send the request
	resp, err := unblocker.MakeRequest(context.Background(), req)
	if err != nil {
		log.Fatal("Error making request:", err)
	}
	defer resp.Body.Close()

	// Read the response body
	body, err := io.ReadAll(resp.Body)
	if err != nil {
		log.Fatal("Error reading response:", err)
	}

	utils.Log("API response successfully received [\u2713]")

	var result APIResponse 
	err = json.Unmarshal(body, &result)
	if err != nil {
		log.Fatal("Error unmarshalling response:", err)
	}

	utils.Log("Reponse body successfully parsed [\u2713]")

	return &result, nil
}

package gradcon

func GradConnectToMPLOY(gradConnect GradConnectData, url string) ([]models.Listing, error) {
	campaigns := gradConnect.Campaigns
	listings := []models.Listing{}

	if len(campaigns) > 0 {
		for _, campaign := range campaigns {
			
			locations, err := GradConLocationsToMPLOYLocationTypes(campaign)
			if err != nil {
				return nil, fmt.Errorf("error getting locations: %v", err)
			}
			
			studyFields, err := GradConStudyFieldsToMPLOYStudyFields(campaign)
			if err != nil {
				return nil, fmt.Errorf("error getting study fields: %v", err)
			}

			jobType, err := GradConJobTypeToMPLOYJobType(campaign)
			
			closeDate := GetGradConCloseDate(campaign)
			startDate := GetGradConStartDate(campaign)
		
			workingRights, err := GradConWorkRightsToMPLOYWorkingRights(campaign)
			if err != nil {
				return nil, fmt.Errorf("error getting working rights: %v", err)
			}

			employerUrl := url + "/employers/" + gradConnect.CustomerOrganisation.Slug
			jobUrl := employerUrl + "/jobs/" + campaign.Slug

			listing := models.Listing{
				Title: campaign.Title,
				Description: campaign.Description,
				Company: models.Company{
					Name: gradConnect.CustomerOrganisation.Name,
					Website: employerUrl,
				},
				ApplicationURL: campaign.OriginTargetURL,
				SourceURLs:[]string{jobUrl},
				Type: jobType,
				CloseDate: &closeDate,
				Locations: locations,
				StudyFields: studyFields,
				StartDate: startDate,
				WorkingRights: workingRights,
				CreatedAt:      time.Now(),
				UpdatedAt:      time.Now(),
			}

			listings = append(listings, listing)
		}
	}

	return listings, nil
}

func GetGradConStartDate(campaign GradConnectCampaign) (time.Time) {
	defaultDate := time.Unix(0, 0).UTC() // Default to 1970-01-01T00:00:00Z
	for _, field := range campaign.AdditionalFields {
		if (field.Type == "start_date") {
			
			valueStr, ok := field.Value.(string)
			if !ok {
				fmt.Printf("warn: error parsing start date: value is not a string, defaulting to 1970-01-01T00:00:00Z\n")
				return defaultDate
			}

			startDate, err := time.Parse("2006-01-02", valueStr)

			if err != nil {
				fmt.Printf("warn: error parsing start date: %v, defaulting to 1970-01-01T00:00:00Z\n", err)

				return defaultDate
			} else {
				return startDate
			} 
		}
	}

	fmt.Printf("warn: error parsing start date: No start date provided, defaulting to 1970-01-01T00:00:00Z\n")
	return defaultDate
}

func GetGradConCloseDate(campaign GradConnectCampaign) (time.Time) {
	defaultDate := time.Unix(0, 0).UTC() // Default to 1970-01-01T00:00:00Z
	endDate := campaign.Interval.End

	parsedEndDate, err := time.Parse(time.RFC3339, endDate)
		
	if err != nil {
		fmt.Printf("warn: error parsing close date: %v, defaulting to 1970-01-01T00:00:00Z\n", err)

		return defaultDate
	} else {
		return parsedEndDate
	}
}

func OpportunityToModel(baseURL string, opportunity Opportunity) (*models.Listing, error) {
	sourceUrl := baseURL + opportunity.DetailPageURl

	company := models.Company{
		Name:    opportunity.ParentEmployer.Title,
		Website: baseURL + opportunity.ParentEmployer.DetailPageUrl,
	}

	jobType, err:= ProspleJobTypeToMPLOYJobType(opportunity.OpportunityTypes[0].Label) // TO BE IMPLEMENTED IN TICKET (MAC-63) 

	parsedCloseDate, err := time.Parse("2006-01-02T15:04:05.000Z", opportunity.ApplicationsCloseDate)
	if err != nil {
		fmt.Printf("warn: error parsing close date: %v, defaulting to 1970-01-01T00:00:00Z\n", err)
		parsedCloseDate = time.Unix(0, 0).UTC() // Default to 1970-01-01T00:00:00Z
	}

	locations, err := ProspleLocationsToMPLOYLocationTypes(opportunity)
	if err != nil {
		return nil, fmt.Errorf("error getting locations: %v", err)
	}

	studyfield, err := ProspleStudyFieldsToMPLOYStudyFields(opportunity)
	if err != nil {
		return nil, fmt.Errorf("error getting study fields: %v", err)
	}

	var parsedStartDate time.Time
	if opportunity.StartDate.Category != nil {
		parsedStartDate, err = time.Parse("2006-01-02T15:04:05.000Z", opportunity.StartDate.Category.Label)

			parsedStartDate = time.Unix(0, 0).UTC() // Default to 1970-01-01T00:00:00Z

	workingRights, err := ProspleWorkingRightsToMPLOYWorkingRights(opportunity)
	if err != nil {
		return nil, fmt.Errorf("error getting working rights: %v", err)

func SendGraphQLQuery(url string, query string, variables map[string]interface{}, unblocker unblocker.Service) (*GraphQLResponse, error) {
	requestBody := GraphQLRequest{
		Query:     query,
		Variables: variables,
	}

	utils.Log("Sending GraphQL request [...]")

	jsonBody, err := json.Marshal(requestBody)
	if err != nil {
		utils.Log("Failed to marshal request body: %v", err)
		return nil, fmt.Errorf("failed to marshal request body: %v", err)
	}

	req, err := http.NewRequest("POST", url, bytes.NewBuffer(jsonBody))
	if err != nil {
		utils.Log("Failed to create request: %v", err)
		return nil, fmt.Errorf("failed to create request: %v", err)
	}

	req.Header.Set("Content-Type", "application/json")

	// Use unblocker instead of http.Client
	resp, err := unblocker.MakeRequest(context.Background(), req)
	if err != nil {
		return nil, fmt.Errorf("failed to send request: %v", err)
	}
	defer resp.Body.Close()

	body, err := io.ReadAll(resp.Body)
	if err != nil {
		return nil, fmt.Errorf("failed to read response body: %v", err)
	}

	if resp.StatusCode != http.StatusOK {
		return nil, fmt.Errorf("HTTP error: %s, %v", resp.Status, string(body))
	}

	utils.Log("GraphQL response successfully received [\u2713]")

	var graphqlResp GraphQLResponse
	if err := json.Unmarshal(body, &graphqlResp); err != nil {
		return nil, fmt.Errorf("failed to unmarshal response: %v", err)
	}

	if len(graphqlResp.Errors) > 0 {
		return nil, fmt.Errorf("GraphQL error: %v", graphqlResp.Errors)
	}

	utils.Log("Reponse body successfully parsed [\u2713]")

	return &graphqlResp, nil
}

// TestCase represents a single test scenario for a scraper
type TestCase struct {
	Name           string
	FixtureFolder  string
	ExpectedFile   string // JSON file containing expected listings
	ExpectError    bool
	FixtureMapping map[string]string // Maps URLs to fixture files
}

// ScraperTestSuite provides common testing functionality for scrapers
type ScraperTestSuite struct {
	T        *testing.T
	Scraper  scraper.Scraper
	TestData string // Base path for test data
}

// mockTransport implements http.RoundTripper for serving local test files
type mockTransport struct {
	t              *testing.T
	fixtureMapping map[string]string
	basePath       string
}

func (m *mockTransport) RoundTrip(req *http.Request) (*http.Response, error) {
	// Get the fixture file for this URL
	fixtureFile, exists := m.fixtureMapping[req.URL.String()]
	if !exists {
		m.t.Fatalf("No fixture mapping found for URL: %s", req.URL.String())
	}

	// Load the fixture
	filename := filepath.Join(m.basePath, fixtureFile)
	content, err := os.ReadFile(filename)
	if err != nil {
		m.t.Fatalf("Failed to read test fixture %s: %v", filename, err)
	}

	// Create mock response
	return &http.Response{
		StatusCode: 200,
		Body:       io.NopCloser(bytes.NewReader(content)),
		Header:     make(http.Header),
	}, nil
}

// NewScraperTestSuite creates a new test suite for a scraper
func NewScraperTestSuite(t *testing.T, scraper scraper.Scraper, testDataPath string) *ScraperTestSuite {
	return &ScraperTestSuite{
		T:        t,
		Scraper:  scraper,
		TestData: testDataPath,
	}
}

// RunTestCase executes a single test case
func (s *ScraperTestSuite) RunTestCase(tc TestCase) {
	s.T.Helper()
	s.T.Run(tc.Name, func(t *testing.T) {
		// Create a new collector with mocked transport
		mockCollector := s.createMockCollector(tc.FixtureMapping, tc.FixtureFolder)

		// Replace the scraper's collector with our mock
		if baseScraper, ok := s.Scraper.(interface{ SetCollector(*colly.Collector) }); ok {
			baseScraper.SetCollector(mockCollector)
		} else {
			t.Fatal("Scraper doesn't implement SetCollector method")
		}

		// Run the scraper
		got, err := s.Scraper.Scrape(context.Background())

		// Check error expectation
		if (err != nil) != tc.ExpectError {
			t.Errorf("Scrape() error = %v, expectError %v", err, tc.ExpectError)
			return
		}

		if !tc.ExpectError {
			// Load expected results
			want := s.loadExpectedListings(tc.ExpectedFile)
			s.compareResults(t, got, want)
		}
	})
}

// createMockCollector creates a collector that uses local test files
func (s *ScraperTestSuite) createMockCollector(fixtureMapping map[string]string, fixturePath string) *colly.Collector {
	c := colly.NewCollector()
	c.WithTransport(&mockTransport{
		t:              s.T,
		fixtureMapping: fixtureMapping,
		basePath:       filepath.Join(s.TestData, fixturePath),
	})
	return c
}

// loadExpectedListings loads expected listings from a JSON file
func (s *ScraperTestSuite) loadExpectedListings(filename string) []models.Listing {
	path := filepath.Join(s.TestData, filename)
	data, err := os.ReadFile(path)
	if err != nil {
		s.T.Fatalf("Failed to read expected listings file: %v", err)
	}

	var listings []models.Listing
	if err := json.Unmarshal(data, &listings); err != nil {
		s.T.Fatalf("Failed to unmarshal expected listings: %v", err)
	}

	return listings
}

// compareResults compares scraped listings with expected listings
func (s *ScraperTestSuite) compareResults(t *testing.T, got, want []models.Listing) {
	t.Helper()

	if len(got) != len(want) {
		t.Errorf("Got %d listings, want %d", len(got), len(want))
		return
	}

	for i := range got {
		s.compareListings(t, got[i], want[i])
	}
}

// compareListings compares two listings for equality
func (s *ScraperTestSuite) compareListings(t *testing.T, got, want models.Listing) {
	t.Helper()

	gotJSON, err := json.Marshal(got)
	if err != nil {
		t.Fatalf("Failed to marshal got listing: %v", err)
	}

	wantJSON, err := json.Marshal(want)
	if err != nil {
		t.Fatalf("Failed to marshal want listing: %v", err)
	}

	if string(gotJSON) != string(wantJSON) {
		t.Errorf("Listings don't match:\ngot:  %s\nwant: %s", gotJSON, wantJSON)
	}
}
