package pipeline

import (
	"context"
	"fmt"
	"github.com/monashcoding/mploy-golang-core/internal/models"
	"github.com/monashcoding/mploy-golang-core/internal/utils"
)

// Step represents a single processing step in the pipeline
type Step interface {
	Name() string
	Process(ctx context.Context, listings []models.Listing) ([]models.Listing, error)
}

// Pipeline manages the processing steps
type Pipeline struct {
	steps []Step
}

func NewPipeline(steps ...Step) *Pipeline {
	return &Pipeline{
		steps: steps,
	}
}

// Run executes all pipeline steps in order
func (p *Pipeline) Run(ctx context.Context, listings []models.Listing) ([]models.Listing, error) {
	var err error
	currentListings := listings

	for _, step := range p.steps {
		utils.Log("Running step: " + step.Name())
		currentListings, err = step.Process(ctx, currentListings)
		if err != nil {
			return nil, fmt.Errorf("error in %s step: %w", step.Name(), err)
		}
		utils.Log("Finished step: " + step.Name())
	}

	return currentListings, nil
}

package steps

import (
	"context"
	"github.com/monashcoding/mploy-golang-core/internal/models"
	"github.com/monashcoding/mploy-golang-core/internal/utils"
	"strings"
)

type LogStep struct {
	prefix string
}

func NewLogStep(prefix string) *LogStep {
	return &LogStep{
		prefix: prefix,
	}
}

func (s *LogStep) Name() string {
	return "Log"
}

func (s *LogStep) Process(ctx context.Context, listings []models.Listing) ([]models.Listing, error) {
	utils.Log("\n=== %s: Processing %d listings ===", s.prefix, len(listings))

	for _, listing := range listings {
		// Create a compact location string
		locations := strings.Join(compactStringSlice(locationsToStrings(listing.Locations)), "/")

		// Get company name or "Unknown"
		company := listing.Company.Name
		if company == "" {
			company = "Unknown"
		}

		// Format close date if exists
		closeDate := "No close date"
		if listing.CloseDate != nil {
			closeDate = listing.CloseDate.Format("2006-01-02")
		}

		// Create compact string for study fields
		fields := strings.Join(compactStringSlice(studyFieldsToStrings(listing.StudyFields)), "/")
		if fields == "" {
			fields = "Any"
		}

		// Print compact single-line summary
		utils.Log("[%s] %s @ %s | Type: %s | Loc: %s | Fields: %s | Closes: %s | URL: %s",
			listing.ID.Hex()[:8],
			truncateString(listing.Title, 40),
			company,
			listing.Type,
			locations,
			fields,
			closeDate,
			truncateString(listing.ApplicationURL, 40),
		)
	}
	return listings, nil
}

// Helper functions

func truncateString(s string, length int) string {
	if len(s) <= length {
		return s
	}
	return s[:length-3] + "..."
}

func compactStringSlice(slice []string) []string {
	if len(slice) <= 3 {
		return slice
	}
	return append(slice[:2], "...")
}

func locationsToStrings(locs []models.LocationType) []string {
	result := make([]string, len(locs))
	for i, loc := range locs {
		result[i] = string(loc)
	}
	return result
}

func studyFieldsToStrings(fields []models.StudyField) []string {
	result := make([]string, len(fields))
	for i, field := range fields {
		result[i] = string(field)
	}
	return result
}

package steps

import (
	"context"

	"github.com/monashcoding/mploy-golang-core/internal/database"
	"github.com/monashcoding/mploy-golang-core/internal/models"
	"github.com/monashcoding/mploy-golang-core/internal/utils"
)

type UploadStep struct {
	mongoDb *database.MongoDB
}

func NewMongoDBStep(mongoDb *database.MongoDB) *UploadStep {
	return &UploadStep{mongoDb}
}

func (s *UploadStep) Name() string {
	return "Upload"
}

func (s *UploadStep) Process(ctx context.Context, listings []models.Listing) ([]models.Listing, error) {
	/* ---------------------------------------------------------- /
	| Naive sync strategy [To be replaced after MVP]        		  |
	| 1. Delete all listings from DB                              |
	| 2. Insert all listings from pipeline                        |
	| 						                                                |	
	| In future, need to think about a smarter way to handle      |
	| updates and deletions. (How do we know a listing is expired |
	| or updated?)                                                |
	| ---------------------------------------------------------- */

	if len(listings) == 0 {
		return listings, nil
	}

	// Sync with database
	utils.Log("Starting database sync [...]")

	utils.Log("Database sync step 1: Deleting all listings [...]")

	err := s.mongoDb.DeleteListings() // Delete all listings
	if err != nil {
		utils.Error("[X] Failed to delete listings: %v", err)
	}
	utils.Log("Database sync step 1: Successfully deleted all listings [\u2713]")

	utils.Log("Database sync step 2: Inserting all listings [...]")
	err = s.mongoDb.CreateListings(listings) // Insert all listings
	if err != nil {
		utils.Error("[X] Failed to insert listings: %v", err)
	}
	utils.Log("Database sync step 2: Successfully inserted all listings [\u2713]")

	dbListings, err := s.mongoDb.GetListings()
	if err != nil {
		utils.Error("[X] Failed to get listings from database: %v", err)
	}

	utils.Log("Database Contains %d Listings", len(dbListings))
	
	return dbListings, nil
}

package testing

import (
	"context"
	"encoding/json"
	"github.com/monashcoding/mploy-golang-core/internal/models"
	"github.com/monashcoding/mploy-golang-core/internal/pipeline"
	"os"
	"path/filepath"
	"testing"
)

// TestCase represents a single test scenario for a pipeline step
type TestCase struct {
	Name        string
	InputFile   string // JSON file containing input listings
	OutputFile  string // JSON file containing expected output listings
	ExpectError bool
}

// PipelineStepTestSuite provides common testing functionality for pipeline steps
type PipelineStepTestSuite struct {
	T        *testing.T
	Step     pipeline.Step
	TestData string // Base path for test data
}

// NewPipelineStepTestSuite creates a new test suite for a pipeline step
func NewPipelineStepTestSuite(t *testing.T, step pipeline.Step, testDataPath string) *PipelineStepTestSuite {
	return &PipelineStepTestSuite{
		T:        t,
		Step:     step,
		TestData: testDataPath,
	}
}

// loadListings loads listings from a JSON file
func (s *PipelineStepTestSuite) loadListings(filename string) []models.Listing {
	path := filepath.Join(s.TestData, filename)
	data, err := os.ReadFile(path)
	if err != nil {
		s.T.Fatalf("Failed to read listings file %s: %v", filename, err)
	}

	var listings []models.Listing
	if err := json.Unmarshal(data, &listings); err != nil {
		s.T.Fatalf("Failed to unmarshal listings from %s: %v", filename, err)
	}

	return listings
}

// compareListings compares two slices of listings
func (s *PipelineStepTestSuite) compareListings(t *testing.T, got, want []models.Listing) {
	t.Helper()

	if len(got) != len(want) {
		t.Errorf("Got %d listings, want %d", len(got), len(want))
		return
	}

	for i := range got {
		gotJSON, err := json.Marshal(got[i])
		if err != nil {
			t.Fatalf("Failed to marshal got listing: %v", err)
		}

		wantJSON, err := json.Marshal(want[i])
		if err != nil {
			t.Fatalf("Failed to marshal want listing: %v", err)
		}

		if string(gotJSON) != string(wantJSON) {
			t.Errorf("Listing %d doesn't match:\ngot:  %s\nwant: %s", i, gotJSON, wantJSON)
		}
	}
}

// RunTestCase executes a single test case
func (s *PipelineStepTestSuite) RunTestCase(tc TestCase) {
	s.T.Helper()
	s.T.Run(tc.Name, func(t *testing.T) {
		// Load input listings
		input := s.loadListings(tc.InputFile)

		// Run the step
		got, err := s.Step.Process(context.Background(), input)

		// Check error expectation
		if (err != nil) != tc.ExpectError {
			t.Errorf("Process() error = %v, expectError %v", err, tc.ExpectError)
			return
		}

		if !tc.ExpectError {
			// Load and compare expected output
			want := s.loadListings(tc.OutputFile)
			s.compareListings(t, got, want)
		}
	})
}
